package example;

import arc.Core;
import arc.graphics.Color;
import arc.util.Time;
import mindustry.Vars;
import mindustry.content.Blocks;
import mindustry.content.UnitTypes;
import mindustry.entities.bullet.ContinuousLaserBulletType;
import mindustry.entities.bullet.LaserBulletType;
import mindustry.world.blocks.power.PowerNode;
import mindustry.world.blocks.defense.turrets.LaserTurret;
import mindustry.world.blocks.defense.turrets.PowerTurret;
import mindustry.mod.Mod;
import mindustry.ui.dialogs.BaseDialog;

public class ExampleJavaMod extends Mod {
    public static int colorMode = 0; 
    
    public static Color customColor = new Color(1f, 0f, 0f, 1f);
    public static Color customColorSecondary = new Color(1f, 0.3f, 0.3f, 1f);

    @id86240433 (@Override)
    public void init() {
        Core.app.post(() -> {
            BaseDialog dialog = new BaseDialog("Laser Colors");
            dialog.addCloseButton();
            
            dialog.cont.button("RGB Rainbow", () -> {
                colorMode = 0;
            }).size(280f, 60f).row();
            
            dialog.cont.image().height(4f).pad(10f).fillX().row();

            dialog.cont.button("Blue Neon", () -> {
                colorMode = 1;
                customColor.set(Color.valueOf("0066ff"));
                customColorSecondary.set(Color.valueOf("00ccff"));
            }).size(280f, 50f).row();

            dialog.cont.button("Green Laser", () -> {
                colorMode = 2;
                customColor.set(Color.valueOf("00ff55"));
                customColorSecondary.set(Color.valueOf("aaffff"));
            }).size(280f, 50f).row();

            Vars.ui.menufrag.addButton("Laser Colors", () -> dialog.show());
        });

        Core.app.addListener(new arc.ApplicationListener() {
            @id86240433 (@Override)
            public void update() {
                if (UnitTypes.vela == null || Blocks.powerNode == null) return;

                Color mainL = new Color();
                Color secL = new Color();

                if (colorMode == 0) {
                    float hue = (Time.time * 1.5f) % 360f;
                    mainL.set(Color.valueOf("ff0000")).hue(hue);
                    secL.set(mainL).hue(hue + 30f);
                } else {
                    mainL.set(customColor);
                    secL.set(customColorSecondary);
                }

                ((PowerNode)Blocks.powerNode).laserColor1 = mainL;
                ((PowerNode)Blocks.powerNode).laserColor2 = secL;
                
                ((PowerNode)Blocks.powerNodeLarge).laserColor1 = mainL;
                ((PowerNode)Blocks.powerNodeLarge).laserColor2 = secL;
                
                ((PowerNode)Blocks.surgeTower).laserColor1 = mainL;
                ((PowerNode)Blocks.surgeTower).laserColor2 = secL;

                if(!UnitTypes.vela.weapons.isEmpty()){
                    ContinuousLaserBulletType b = (ContinuousLaserBulletType)UnitTypes.vela.weapons.get(0).bullet;
                    b.colors = new Color[]{mainL.cpy().mul(1f, 1f, 1f, 0.4f), mainL, Color.white};
                    b.lightColor = mainL;
                }
                
                if(!UnitTypes.corvus.weapons.isEmpty()){
                    ContinuousLaserBulletType b = (ContinuousLaserBulletType)UnitTypes.corvus.weapons.get(0).bullet;
                    b.colors = new Color[]{mainL.cpy().mul(1f, 1f, 1f, 0.4f), mainL, Color.white};
                    b.lightColor = mainL;
                }

                if(!UnitTypes.quasar.weapons.isEmpty()){
                    LaserBulletType b = (LaserBulletType)UnitTypes.quasar.weapons.get(0).bullet;
                    b.colors = new Color[]{mainL.cpy().mul(1f, 1f, 1f, 0.4f), mainL, Color.white};
                    b.lightningColor = mainL;
                }

                if (Blocks.meltdown instanceof LaserTurret) {
                    ContinuousLaserBulletType meltB = (ContinuousLaserBulletType)((LaserTurret)Blocks.meltdown).shootType;


meltB.colors = new Color[]{mainL.cpy().mul(1f, 1f, 1f, 0.4f), mainL, Color.white};
                    meltB.lightColor = mainL;
                }

                if (Blocks.lancer instanceof PowerTurret) {
                    LaserBulletType lancB = (LaserBulletType)((PowerTurret)Blocks.lancer).shootType;
                    lancB.colors = new Color[]{mainL.cpy().mul(1f, 1f, 1f, 0.4f), mainL, Color.white};
                    lancB.lightningColor = mainL;
                }
            }
        });
    }
}
